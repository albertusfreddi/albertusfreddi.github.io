Interactive Escalation Button Dispatcher
import httpx

async def send_ops_alert(bot_token: str, chat_id: str, incident: dict):
    url = f"https://api.telegram.org/bot{bot_token}/sendMessage"
    payload = {
        "chat_id": chat_id,
        "text": f"🔥 *CRITICAL INCIDENT* \nService: `{incident['service']}`\nError: {incident['error']}",
        "parse_mode": "Markdown",
        "reply_markup": {
            "inline_keyboard": [[
                {"text": "✅ Acknowledge", "callback_data": f"ack_{incident['id']}"},
                {"text": "🚨 Escalate Lead", "callback_data": f"esc_{incident['id']}"}
            ]]
        }
    }
    async with httpx.AsyncClient() as client:
        await client.post(url, json=payload, timeout=5.0)
