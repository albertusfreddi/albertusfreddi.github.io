async def fetch_target_resource(target_url: str, proxy_address: str):
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
        "Accept": "application/json, text/plain, */*",
        "Accept-Language": "id-ID,id;q=0.9,en-US;q=0.8"
    }
    # Impersonate authentic modern browser TLS fingerprint
    resp = requests.get(target_url, headers=headers, impersonate="chrome120", proxies={"https": proxy_address}, timeout=10)
    if resp.status_code == 200:
        return resp.json()
    raise RuntimeError(f"Rate limited or challenged: HTTP {resp.status_code}")
